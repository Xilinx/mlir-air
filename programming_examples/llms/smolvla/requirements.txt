# Additional Python packages for the SmolVLA example.
#
# Install into the same environment that has air + pyxrt (the mlir-air one):
#
#     pip install -r requirements.txt
#
# `make verify` then runs with the default LEROBOT_PYTHON=python3, the same way
# the sibling examples do.
#
# lerobot pins its own numpy range, which may move numpy in an interpreter
# that every other example shares. To keep the shared environment untouched,
# install into a separate venv and point the Makefile at it; sourcing
# utils/env_setup.sh puts air and pyxrt on PYTHONPATH so they import from any
# venv:
#
#     make verify LEROBOT_PYTHON=/path/to/venv/bin/python
#
# The example downloads lerobot/smolvla_base and its SmolVLM2 backbone on first
# use. Both are public; no HuggingFace token is needed.

# Weight loading
safetensors
huggingface_hub

# The SmolVLA policy itself. This IS the CPU baseline that `make verify`
# compares against: the example splices the NPU vision encoder into this
# unmodified model, so the reference is upstream lerobot rather than a
# reimplementation living in this repo.
#
# The [dataset] extra is NOT optional despite the name: `import
# lerobot.policies` pulls in the GR00T policy, which imports
# lerobot.datasets, which hard-requires `av`. A bare `pip install lerobot`
# fails at import with "'av' is required but not installed".
lerobot[dataset]
torch

# Tokenizer + the SigLIP/SmolVLM config the vision weights are read from.
# num2words is a hard requirement of the SmolVLM processor in transformers
# (raises ImportError in processing_smolvlm.py without it) but is not pulled in
# by transformers itself.
transformers>=5.3.0
num2words

# Host-side reference math and bf16 handling
numpy
ml_dtypes
# Caps numpy's BLAS threads inside the NPU forward, where waking every hardware thread
# slows the host side of the vision stage. Optional: without it the cap is skipped with a
# warning.
threadpoolctl
