# Additional Python packages for the Qwen3-Coder-30B-A3B example.
#
# These are *on top of* the mlir-air base environment (which already provides
# numpy, ml_dtypes, pyxrt, etc.). Install with:
#     pip install -r requirements.txt

# Download the bf16 checkpoint, its tokenizer and config from the Hub at their
# pinned revision, and read its safetensors to quantize them.
huggingface_hub
safetensors
tokenizers

# The quantizer, the CPU reference and the host side of the prefill.
torch

# `make verify` checks the CPU reference against transformers' Qwen3-MoE
# layers and runs the top-k token-set gate against the bf16 model.
transformers

# Serialises the NPU between concurrent runs.
filelock
